const express = require('express'); // load express // to load you need "reqire"
const userRoutes = require('./routes/users'); // Load the user routes
const postRoutes = require('./routes/posts'); // import the post routes
require('dotenv').config(); // Load the package dotenv to use environment variables
const cors = require('cors'); // Load the package cors to handle cross-origin requests
const morgan = require('morgan'); // Load the package morgan for logging HTTP requests



const app = express();
const PORT = process.env.PORT || 5000; // Use the port from the environment variable or default to 5000

app.use(express.json()); // Middleware to parse JSON
app.use(cors(
  {
    origin: ['http://localhost:3000', 'http://localhost:5173']
  }
)); // Middleware to enable CORS

app.use(morgan('combined'))
app.use('/users', userRoutes); // Use the user routes
app.use('/posts', postRoutes); // Use the post routes
